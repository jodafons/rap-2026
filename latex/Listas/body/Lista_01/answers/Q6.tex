def calcular_media_e_desvio_padrao( x1, x2, x3, x4, x5, x6 ):
    media = (x1+x2+x3+x4+x5+x6)/6
    sigma2 = (x1-media)**2
    sigma2+= (x2-media)**2
    sigma2+= (x3-media)**2
    sigma2+= (x4-media)**2
    sigma2+= (x5-media)**2
    sigma2+= (x6-media)**2
    sigma2/=6
    sigma = sigma2**(1/2)
    return media, sigma2, sigma
