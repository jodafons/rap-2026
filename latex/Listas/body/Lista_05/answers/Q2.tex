def produto_escalar(a, b):
    resultado = 0.0 if any(isinstance(x, float) for x in a + b) else 0
    i = 0
    while i < len(a):
        resultado += a[i] * b[i]
        i += 1
    return resultado

# Testes (Item b):

print(produto_escalar([2.5, 0.0], [4.0, 10.0]))    # Esperado: 10.0
print(produto_escalar([], []))                    # Esperado: 0
