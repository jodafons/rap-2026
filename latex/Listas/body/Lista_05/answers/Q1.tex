# Resposta Questão 1:
def busca_posicoes(lista, numero):
    posicoes = []
    i = 0
    while i < len(lista):
        if lista[i] == numero:
            posicoes.append(i)
        i += 1
    return posicoes

# Testes (Item b):
print(busca_posicoes([10, 20, 30, 20, 40, 20], 20)) # Esperado: [1, 3, 5]
print(busca_posicoes([1, 2, 3, 4, 5], 99))            # Esperado: []
