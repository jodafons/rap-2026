def processa_tags(texto_bruto):
    # Utiliza .strip(), .lower(), .replace() e .split()
    texto_sanitizado = texto_bruto.strip().lower().replace("-", " ")
    tags = texto_sanitizado.split(",")
    return [tag.strip() for tag in tags]

# Testes (Item b):
