from typing import Tuple

def analisar_credito_bancario(
    renda_mensal: float,
    parcela: float,
    score: int,
    tem_divida_ativa: bool,
) -> Tuple[bool, str]:
    if tem_divida_ativa:
        return (
            False,
            "RECUSADO_INADIMPLENCIA",
        )
    if parcela > (
        renda_mensal * 0.30
    ):
        return (
            False,
            "RECUSADO_MARGEM",
        )
    if score < 500:
        return (
            False,
            "RECUSADO_SCORE",
        )
    elif score >= 700:
        return (
            True,
            "APROVADO_JUROS_BAIXOS",
        )
    else:
        return (
            True,
            "APROVADO_JUROS_ALTOS",
        )
