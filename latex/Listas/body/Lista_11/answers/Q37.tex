from typing import Tuple

def gerar_fibonacci(
    n_termos: int,
) -> Tuple[int, ...]:
    if n_termos <= 0:
        return ()
    if n_termos == 1:
        return (0,)
    
    seq = [0, 1]
    while len(seq) < n_termos:
        prox = seq[-1] + seq[-2]
        seq.append(prox)
    return tuple(seq)
