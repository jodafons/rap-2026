def validar_primeiro_digito_cpf(
    cpf_str: str,
) -> bool:
    if (
        len(cpf_str) != 11
        or not cpf_str.isdigit()
    ):
        return False
    
    soma = 0
    peso = 10
    for i in range(9):
        soma += int(cpf_str[i]) * peso
        peso -= 1
        
    resto = soma % 11
    d_esp = (
        0 if resto < 2
        else 11 - resto
    )
    return int(cpf_str[9]) == d_esp
