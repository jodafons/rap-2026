\begin{minted}{python}
def calcular_ir_simplificado(
    salario_bruto: float
) -> float:
    if salario_bruto <= 2000.0:
        return 0.0
    elif salario_bruto <= 4000.0:
        excesso = salario_bruto - 2000.0
        return excesso * 0.10
    else:
        excesso = salario_bruto - 4000.0
        return 200.0 + (excesso * 0.20)
\end{minted}
