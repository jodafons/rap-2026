def formatar_catalogo(lista_raw):
    catalogo = []
    for item in lista_raw:
        titulo_bruto, categoria = item.split("-")
        titulo_limpo = titulo_bruto.replace("_", " ")
        catalogo.append((titulo_limpo, categoria))
    return catalogo
