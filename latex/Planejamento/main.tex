\ExplSyntaxOn
\cs_new_protected:Npn \__code_word:n #1 {
  \tl_if_empty:nF { #1 }
   {
     \colorbox{cinza-codigo}{\texttt{#1}}
   }
}

\NewDocumentCommand{\code}{m}
 {
   \seq_set_split:Nnn \l_tmpa_seq { ~ } { #1 }
   \int_set:Nn \l_tmpa_int { 0 }
   \seq_map_inline:Nn \l_tmpa_seq
    {
      \int_incr:N \l_tmpa_int
      \int_compare:nF { \l_tmpa_int == 1 } { \allowbreak\ \hspace{0pt} }
      \tl_set:Nn \l_tmpa_tl { ##1 }
      \tl_replace_all:Nnn \l_tmpa_tl { \_ } { \__code_word:n { \_ } \allowbreak }
      \tl_replace_all:Nnn \l_tmpa_tl { ( } { \__code_word:n { ( } \allowbreak }
      \tl_replace_all:Nnn \l_tmpa_tl { ) } { \__code_word:n { ) } \allowbreak }
      \tl_replace_all:Nnn \l_tmpa_tl { [ } { \__code_word:n { [ } \allowbreak }
      \tl_replace_all:Nnn \l_tmpa_tl { ] } { \__code_word:n { ] } \allowbreak }
      \tl_replace_all:Nnn \l_tmpa_tl { , } { \__code_word:n { , } \allowbreak }
      \tl_replace_all:Nnn \l_tmpa_tl { : } { \__code_word:n { : } \allowbreak }
      \tl_replace_all:Nnn \l_tmpa_tl { / } { \__code_word:n { / } \allowbreak }
      \tl_replace_all:Nnn \l_tmpa_tl { . } { \__code_word:n { . } \allowbreak }
      \__code_word:n { \l_tmpa_tl }
    }
 }
\ExplSyntaxOff
